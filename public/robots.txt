# One group per crawler, and a named group REPLACES the * group entirely — a
# crawler never merges the two. So every rule below has to be repeated in the
# answer-engine group; when only `Allow: /` was listed there, GPTBot and the
# rest were free to crawl /s/ share tokens and every authed duplicate of "/".
# Consecutive User-agent lines share one group, so the four named agents are
# listed together rather than as four identical copies.

User-agent: *
Allow: /
Disallow: /api/
Disallow: /admin
Disallow: /settings
# Authed app surfaces. A signed-out crawler that hits one of these gets the
# landing page rendered back at it (the SPA parent route), so without these it
# would index a dozen duplicates of "/". Nothing here is a real page to a
# logged-out visitor.
Disallow: /echoes
Disallow: /resonance
Disallow: /journal
Disallow: /me
Disallow: /collections/
# Reader surfaces, not marketing. /s/ is a share token — indexing one would
# turn a link meant for a few people into a public page. /u/ needs a signed-in
# viewer to resolve blocks and visibility, so a crawler only ever sees 404.
Disallow: /s/
Disallow: /u/

# Answer engines are a growth channel, not a threat: the genuinely public
# surface is the landing page and the legal pages, and they are welcome to it.
# The disallows are the same list as above, not a laxer one.
User-agent: GPTBot
User-agent: ClaudeBot
User-agent: PerplexityBot
User-agent: Google-Extended
Allow: /
Disallow: /api/
Disallow: /admin
Disallow: /settings
Disallow: /echoes
Disallow: /resonance
Disallow: /journal
Disallow: /me
Disallow: /collections/
Disallow: /s/
Disallow: /u/
